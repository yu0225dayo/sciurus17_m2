================================================================
 パイプライン起動手順
================================================================

【構成】
  ROS PC (Linux, RealSense + ROS1 + ロボット接続)
    └─ ros/rgbd_server.py  ← RGBD を HTTP で提供 + 手首座標を ROS にパブリッシュ
          ↓ HTTP :5000 (RGBD送信 / 手首座標受信)
  Client PC (Windows, GPU なし)
    └─ client/grasp_pipeline.py  ← 姿勢推定依頼 + 把持姿勢生成 + 手首座標送信
          ↓ HTTP :8080
  GPU サーバ (計算機 10.40.1.126)
    ├─ server.py          (SAM-3D + SAM-6D, ポート 8080)
    └─ sam6d Docker       (SAM-6D, ポート 8081, server.py が内部呼び出し)

  データフロー:
    ROS PC → [RGBD] → Client PC → [RGBD] → GPU サーバ
                                          ← [R, t, points]
                                  → GraspGenerator
    ROS PC ← [hand[0] xyz] ← Client PC
    /wrist_left, /wrist_right (geometry_msgs/PointStamped)

================================================================
 STEP 1: GPU サーバ側 — SAM-6D Docker を起動
================================================================

  cd /home/okada/ws/project/server
  docker compose up -d sam6d

  # 起動確認 (モデルロードに1〜2分かかる)
  docker logs -f sam6d_service

  # ヘルスチェック
  curl http://localhost:8081/health
  # → {"status":"ok","models_loaded":true} が返れば OK

================================================================
 STEP 2: サーバ側 — server.py を起動
================================================================

  cd /home/okada/ws/project/server
  python server.py \
      --sam-checkpoint /ws/okada/project/sam_vit_h_4b8939.pth \
      --sam3d-config   ~/ws/sam-3d-objects/checkpoints/hf/pipeline.yaml \
      --sam3d-repo     ~/ws/sam-3d-objects \
      --sam6d-service  http://localhost:8081 \
      --host 0.0.0.0 --port 8080

  # バックグラウンドで実行する場合
  nohup python server.py \
      --sam-checkpoint /ws/okada/project/sam_vit_h_4b8939.pth \
      --sam3d-config   ~/ws/sam-3d-objects/checkpoints/hf/pipeline.yaml \
      --sam3d-repo     ~/ws/sam-3d-objects \
      --sam6d-service  http://localhost:8081 \
      --host 0.0.0.0 --port 8080 \
      > server.log 2>&1 &

================================================================
 STEP 3: ROS PC 側 — RGBD HTTP サーバを起動
================================================================

  # 依存インストール (初回のみ)
  pip install flask pyrealsense2 opencv-python numpy

  # 起動 (ROS 環境に source してから)
  source /opt/ros/noetic/setup.bash
  cd <このリポジトリ>/ros
  python rgbd_server.py --port 5000

  # ヘルスチェック
  curl http://localhost:5000/health
  # → {"camera_ready": true, "status": "ok"} が返れば OK

  # パブリッシュされる ROS トピック
  rostopic echo /wrist_left   # geometry_msgs/PointStamped
  rostopic echo /wrist_right

================================================================
 STEP 4: Client PC 側 — 把持パイプライン実行
================================================================

  cd <このリポジトリ>/client

  # 実行 (物体の中心付近をクリック座標で指定)
  python grasp_pipeline.py \
      --ros-pc     http://<ROS_PC_IP>:5000 \
      --gpu-server http://10.40.1.126:8080 \
      --click-x 320 --click-y 240

  # オプション説明
  #   --ros-pc      ROS PC の IP アドレスとポート  [必須]
  #   --gpu-server  GPU サーバの URL              [デフォルト: http://10.40.1.126:8080]
  #   --click-x/y   物体クリック座標               [-1 で未指定 (画像中央を使用)]
  #   --num-samples 生成する把持候補数              [デフォルト: 1]
  #   --epoch       PositionVAE エポック番号       [デフォルト: 69]
  #   --model-dir   Shape2Gesture モデルディレクトリ [デフォルト: save_model]

================================================================
 出力 (ROS PC 側 ROS トピック)
================================================================

  /wrist_left   geometry_msgs/PointStamped  左手首位置 (カメラ座標系 [m])
  /wrist_right  geometry_msgs/PointStamped  右手首位置 (カメラ座標系 [m])

  header.frame_id = "camera_color_optical_frame"
